import { tanstackStart } from '@tanstack/react-start/plugin/vite';
import viteReact from '@vitejs/plugin-react';
import { defineConfig } from 'vite';
import path from 'node:path';
const root = process.cwd();
export default defineConfig({
 root: path.join(root, 'apps/web'),
 resolve: { tsconfigPaths: true },
 plugins: [{name:'disposable-local-boundaries', enforce:'pre',
 configureServer(server) { server.middlewares.use('/__test/reset', (_req,res)=>{globalThis.__liveReset=true;res.end('reset');}); },
 transform(code,id) {
  if(id.endsWith('/server/workspace-session.ts')) {
   code=code.replace('const connection = await databaseUrl();', 'const connection = "local-only";');
   code=code.replace('const database = openPostgres(connection);', `
    if (!globalThis.__liveDatabase) globalThis.__liveDatabase = (async () => {
     const { openPglite } = await import('${root}/packages/adapters/test/pglite.ts');
     const db = await openPglite();
     const { replaceDemoAccount } = await import('${root}/packages/seed/src/demo-account.ts');
     await replaceDemoAccount(db.db, 'live-test', 'live-user');
     return db;
    })();
    const local = await globalThis.__liveDatabase;
    if(globalThis.__liveReset) {
     globalThis.__liveReset=false;
     const { replaceDemoAccount } = await import('${root}/packages/seed/src/demo-account.ts');
     await replaceDemoAccount(local.db, 'live-test', 'live-user');
    }
    const database = {db:local.db, close:async()=>{}};
   `);
   return code;
  }
  if(id.endsWith('/server/session.ts')) return 'export async function requireClerkUser(){return "live-user";}';
 }
 }, tanstackStart(), viteReact()],
 server: {host:'127.0.0.1',port:3000},
});
